import { NextRequest, NextResponse } from 'next/server'
import { parseDocuSealWebhook } from '@/lib/docuseal'
import { createAdminClient } from '@/lib/supabase-server'

export async function POST(request: NextRequest) {
  const body = await request.json()
  const { submissionId, isComplete, documentUrl } = parseDocuSealWebhook(body)

  if (!isComplete) {
    // Partial signing â€” one party signed but not all. Log and wait.
    console.log(`[DocuSeal] Submission ${submissionId} not yet complete:`, body.data?.status)
    return NextResponse.json({ received: true })
  }

  const supabase = createAdminClient()

  // Find the SHE file linked to this DocuSeal submission
  const { data: file, error } = await supabase
    .from('she_files')
    .select('id, contractor_id, status')
    .eq('docuseal_submission_id', submissionId)
    .single()

  if (error || !file) {
    console.error('[DocuSeal] Could not find SHE file for submission:', submissionId)
    return NextResponse.json({ error: 'File not found' }, { status: 404 })
  }

  // Update file status to completed and store the signed document URL
  const { error: updateError } = await supabase
    .from('she_files')
    .update({
      status: 'completed',
      docuseal_status: 'completed',
      docuseal_document_url: documentUrl,
      completed_at: new Date().toISOString(),
      updated_at: new Date().toISOString(),
    })
    .eq('id', file.id)

  if (updateError) {
    console.error('[DocuSeal] Failed to update file status:', updateError)
    return NextResponse.json({ error: 'Update failed' }, { status: 500 })
  }

  // Update consultant's completed file count
  const { data: sheFile } = await supabase
    .from('she_files')
    .select('consultant_id')
    .eq('id', file.id)
    .single()

  if (sheFile?.consultant_id) {
    await supabase.rpc('increment_consultant_files', {
      consultant_uuid: sheFile.consultant_id
    })
    // SQL for this RPC:
    // CREATE OR REPLACE FUNCTION increment_consultant_files(consultant_uuid UUID)
    // RETURNS void AS $$
    //   UPDATE consultants
    //   SET files_completed = files_completed + 1,
    //       files_this_month = files_this_month + 1
    //   WHERE id = consultant_uuid;
    // $$ LANGUAGE sql;
  }

  console.log(`[DocuSeal] SHE file ${file.id} completed. Document URL: ${documentUrl}`)
  return NextResponse.json({ received: true })
}

// Creates a new SHE file after the contractor completes the intake wizard

import { NextRequest, NextResponse } from 'next/server'
import { createServerSupabaseClient } from '@/lib/supabase-server'
import { buildPayFastPayment, } from '@/lib/payfast'
import { getFilePrice } from '@/lib/pricing'
import { z } from 'zod'

const CreateFileSchema = z.object({
  project_name:         z.string().min(3),
  project_address:      z.string().min(5),
  project_value:        z.number().positive().optional(),
  project_start_date:   z.string().optional(),
  project_end_date:     z.string().optional(),
  principal_contractor: z.string().optional(),
  client_name:          z.string().optional(),
  site_agent:           z.string().optional(),
  volume_tier:          z.enum(['standard', 'volume_a', 'volume_b', 'volume_c']).default('standard'),
  // Section content from wizard
  sections:             z.record(z.unknown()).optional(),
})

export async function POST(request: NextRequest) {
  const supabase = createServerSupabaseClient()

  // 1. Verify the user is authenticated
  const { data: { user }, error: authError } = await supabase.auth.getUser()
  if (authError || !user) {
    return NextResponse.json({ error: 'Unauthorised' }, { status: 401 })
  }

  // 2. Verify they are a contractor
  const { data: profile } = await supabase
    .from('profiles')
    .select('role, full_name')
    .eq('id', user.id)
    .single()

  if (profile?.role !== 'contractor') {
    return NextResponse.json({ error: 'Only contractors can create SHE files' }, { status: 403 })
  }

  // 3. Parse and validate body
  const body = await request.json()
  const parsed = CreateFileSchema.safeParse(body)
  if (!parsed.success) {
    return NextResponse.json({ error: parsed.error.flatten() }, { status: 422 })
  }

  const { sections, volume_tier, ...fileData } = parsed.data
  const priceZar = getFilePrice(volume_tier)

  // 4. Insert the SHE file (status: 'draft' â†’ will become 'payment_pending')
  const { data: file, error: insertError } = await supabase
    .from('she_files')
    .insert({
      contractor_id: user.id,
      volume_tier,
      price_zar: priceZar,
      status: 'payment_pending',
      ...fileData,
    })
    .select()
    .single()

  if (insertError || !file) {
    console.error('[files/create] Insert error:', insertError)
    return NextResponse.json({ error: 'Failed to create file' }, { status: 500 })
  }

  // 5. Optionally insert wizard section data
  if (sections && Object.keys(sections).length > 0) {
    const sectionRows = Object.entries(sections).map(([key, content]) => ({
      she_file_id: file.id,
      section_key: key,
      content,
    }))
    await supabase.from('she_file_sections').insert(sectionRows)
  }

  // 6. Build PayFast payment object
  const payment = buildPayFastPayment({
    sheFileId:        file.id,
    fileNumber:       file.file_number,
    contractorEmail:  user.email!,
    contractorName:   profile.full_name,
    amountCents:      priceZar,
  })

  return NextResponse.json({
    file,
    payment,   // front end uses this to redirect to PayFast
  }, { status: 201 })
}

// PayFast Instant Transaction Notification (ITN) webhook
// PayFast POSTs here when a contractor payment completes or fails

import { NextRequest, NextResponse } from 'next/server'
import { validateITN } from '@/lib/payfast'
import { createAdminClient } from '@/lib/supabase-server'
import { splitRevenue } from '@/lib/pricing'

export async function POST(request: NextRequest) {
  const formData = await request.formData()
  const body = Object.fromEntries(formData.entries()) as Record<string, string>

  // 1. Validate the ITN is genuinely from PayFast
  const isValid = await validateITN(body, request.headers)
  if (!isValid) {
    console.error('[PayFast ITN] Validation failed', body)
    return new NextResponse('Invalid ITN', { status: 400 })
  }

  const { payment_status, m_payment_id: sheFileId, pf_payment_id } = body

  const supabase = createAdminClient()

  if (payment_status === 'COMPLETE') {
    // 2. Fetch the file to get price and consultant tier
    const { data: file } = await supabase
      .from('she_files')
      .select('*, consultants(tier)')
      .eq('id', sheFileId)
      .single()

    if (!file) {
      console.error('[PayFast ITN] SHE file not found:', sheFileId)
      return new NextResponse('Not found', { status: 404 })
    }

    const consultantTier = (file as any).consultants?.tier ?? 'free'
    const { platformFee, consultantAmount, feePct } = splitRevenue(
      file.price_zar,
      consultantTier
    )

    // 3. Update SHE file status â†’ paid, set 48h SLA deadline
    const slaDeadline = new Date()
    slaDeadline.setHours(slaDeadline.getHours() + 48)

    await supabase.from('she_files').update({
      status: 'paid',
      paid_at: new Date().toISOString(),
      payfast_payment_id: pf_payment_id,
      sla_deadline: slaDeadline.toISOString(),
      updated_at: new Date().toISOString(),
    }).eq('id', sheFileId)

    // 4. Record the transaction
    await supabase.from('transactions').insert({
      she_file_id:          sheFileId,
      contractor_id:        file.contractor_id,
      consultant_id:        file.consultant_id,
      gross_amount:         file.price_zar,
      platform_fee_pct:     feePct,
      platform_fee_amount:  platformFee,
      consultant_amount:    consultantAmount,
      payfast_payment_id:   pf_payment_id,
    })

    console.log(`[PayFast ITN] Payment complete for ${sheFileId}, R${file.price_zar / 100}`)
  } else if (payment_status === 'CANCELLED') {
    await supabase.from('she_files').update({
      status: 'cancelled',
      updated_at: new Date().toISOString(),
    }).eq('id', sheFileId)
  }

  // PayFast expects a 200 OK â€” anything else triggers a retry
  return new NextResponse('OK', { status: 200 })
}

// Supabase magic link / email confirmation callback

import { createServerClient } from '@supabase/ssr'
import { cookies } from 'next/headers'
import { NextRequest, NextResponse } from 'next/server'

export async function GET(request: NextRequest) {
  const { searchParams, origin } = new URL(request.url)
  const code = searchParams.get('code')
  const next = searchParams.get('next') ?? '/'

  if (code) {
    const cookieStore = cookies()
    const supabase = createServerClient(
      process.env.NEXT_PUBLIC_SUPABASE_URL!,
      process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
      {
        cookies: {
          get(name) { return cookieStore.get(name)?.value },
          set(name, value, options) { cookieStore.set({ name, value, ...options }) },
          remove(name, options) { cookieStore.set({ name, value: '', ...options }) },
        },
      }
    )

    const { error } = await supabase.auth.exchangeCodeForSession(code)
    if (!error) {
      const { data: { user } } = await supabase.auth.getUser()
      if (user) {
        const { data: profile } = await supabase
          .from('profiles').select('role').eq('id', user.id).single()
        return NextResponse.redirect(`${origin}/${profile?.role ?? ''}`)
      }
      return NextResponse.redirect(`${origin}${next}`)
    }
  }

  return NextResponse.redirect(`${origin}/auth/login?error=Could not confirm email`)
}

  projectName: string
  projectAddress: string
  consultantEmail: string
  consultantName: string
  contractorEmail: string
  contractorName: string

          'consultant_name': params.consultantName,
          'signing_date':    new Date().toLocaleDateString('en-ZA'),
        },
        metadata: { she_file_id: params.sheFileId }
      },
      {
        email: params.contractorEmail,
        name: params.contractorName,
        role: 'Contractor',         // receives a copy; may not need to sign
        send_email: true,
      }
    ]
  }

  const res = await fetch(`${DOCUSEAL_URL}/submissions`, {
    method: 'POST',
    headers: docusealHeaders(),
    body: JSON.stringify(body),
  })

  if (!res.ok) {
    const err = await res.text()
    throw new Error(`DocuSeal submission creation failed: ${err}`)
  }

  return res.json()

import crypto from 'crypto'

const MERCHANT_ID  = process.env.PAYFAST_MERCHANT_ID!
const MERCHANT_KEY = process.env.PAYFAST_MERCHANT_KEY!

const APP_URL = process.env.NEXT_PUBLIC_APP_URL!

interface PayFastParams {

  contractorName: string

/**
 * Builds the PayFast payment data object and generates the MD5 signature.
 * The returned object can be used to POST to PayFast or render a payment form.
 */
export function buildPayFastPayment(params: PayFastParams) {
  const amountRand = (params.amountCents / 100).toFixed(2)

  const data: Record<string, string> = {
    merchant_id:       MERCHANT_ID,
    merchant_key:      MERCHANT_KEY,
    return_url:        `${APP_URL}/contractor/payment/success?file=${params.sheFileId}`,
    cancel_url:        `${APP_URL}/contractor/payment/cancelled?file=${params.sheFileId}`,
    notify_url:        `${APP_URL}/api/payments/notify`,   // ITN webhook
    name_first:        params.contractorName.split(' ')[0] ?? params.contractorName,
    name_last:         params.contractorName.split(' ').slice(1).join(' ') || '-',
    email_address:     params.contractorEmail,
    m_payment_id:      params.sheFileId,      // our internal ID
    amount:            amountRand,
    item_name:         `ProtekFlow SHE File ${params.fileNumber}`,
    item_description:  'SACPCMP-signed SHE file â€” OHS Act 85/1993',
    custom_str1:       params.sheFileId,
  }

  const signature = generateSignature(data)
  return { ...data, signature, payfast_url: PAYFAST_URL }
}

/** Generates the MD5 signature PayFast requires */
function generateSignature(data: Record<string, string>): string {
  // Build query string in the exact order PayFast expects
  const queryString = Object.entries(data)
    .filter(([, v]) => v !== '')
    .map(([k, v]) => `${k}=${encodeURIComponent(v.trim()).replace(/%20/g, '+')}`)
    .join('&')

  const stringToSign = PASSPHRASE
    ? `${queryString}&passphrase=${encodeURIComponent(PASSPHRASE.trim()).replace(/%20/g, '+')}`
    : queryString

  return crypto.createHash('md5').update(stringToSign).digest('hex')
}

/**
 * Validates a PayFast ITN (Instant Transaction Notification) webhook.
 * Call this in /api/payments/notify BEFORE updating your database.
 *
 * Returns true if the ITN is genuine, false if it should be rejected.
 */
export async function validateITN(
  body: Record<string, string>,
  requestHeaders: Headers
): Promise<boolean> {
  // 1. Verify the IP is a PayFast IP
  const payfastIPs = [
    '197.97.145.144', '197.97.145.145', '197.97.145.146', '197.97.145.147',
    // Sandbox
    '197.97.145.148', '197.97.145.149',
  ]
  const requestIP = requestHeaders.get('x-forwarded-for')?.split(',')[0]?.trim()
  if (requestIP && !payfastIPs.includes(requestIP) && !IS_SANDBOX) {
    console.warn('[PayFast ITN] Rejected: IP not in whitelist:', requestIP)
    return false
  }

  // 2. Verify signature
  const { signature: receivedSig, ...dataWithoutSig } = body
  const expectedSig = generateSignature(dataWithoutSig)
  if (receivedSig !== expectedSig) {
    console.warn('[PayFast ITN] Rejected: signature mismatch')
    return false
  }

  // 3. Verify with PayFast server (pfValidate)
  const pfHost = IS_SANDBOX ? 'sandbox.payfast.co.za' : 'www.payfast.co.za'
  const pfData = Object.entries(body)
    .map(([k, v]) => `${k}=${encodeURIComponent(v.trim())}`)
    .join('&')

  const pfResponse = await fetch(`https://${pfHost}/eng/query/validate`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: pfData,
  })
  const pfValidation = await pfResponse.text()

  if (pfValidation !== 'VALID') {
    console.warn('[PayFast ITN] Rejected: PayFast validation response:', pfValidation)
    return false
  }

  return true
}

/** Splits text into lines that fit within maxWidth */
function wrapText(text: string, font: PDFFont, fontSize: number, maxWidth: number): string[] {
  const words = text.split(' ')
  const lines: string[] = []
  let current = ''

  for (const word of words) {
    const test = current ? `${current} ${word}` : word
    const width = font.widthOfTextAtSize(test, fontSize)
    if (width > maxWidth && current) {
      lines.push(current)
      current = word
    } else {
      current = test
    }
  }
  if (current) lines.push(current)
  return lines
}

// Single source of truth for all ProtekFlow pricing logic

export const PRICING = {
  // File prices in ZAR cents (avoids float bugs)
  FILE_STANDARD:   450000,   // R4,500
  FILE_VOLUME_A:   412500,   // R4,125  (3â€“5 files with same consultant)
  FILE_VOLUME_B:   362500,   // R3,625  (6â€“11 files)
  FILE_VOLUME_C:   312500,   // R3,125  (12+ files)

  // Consultant monthly subscription
  PRO_MONTHLY:     99900,    // R999

  // Platform fee percentages
  FEE_FREE:        25,       // 25% on Free tier
  FEE_PRO:         15,       // 15% on Pro tier

  // SLA
  SLA_HOURS:       48,
} as const

export type VolumeTier = 'standard' | 'volume_a' | 'volume_b' | 'volume_c'
export type ConsultantTier = 'free' | 'pro'

/** Returns the file price in cents for a given volume tier */
export function getFilePrice(tier: VolumeTier = 'standard'): number {
  const map: Record<VolumeTier, number> = {
    standard: PRICING.FILE_STANDARD,
    volume_a: PRICING.FILE_VOLUME_A,
    volume_b: PRICING.FILE_VOLUME_B,
    volume_c: PRICING.FILE_VOLUME_C,
  }
  return map[tier]
}

/** Splits a gross file price into platform fee and consultant amount */
export function splitRevenue(
  grossCents: number,
  consultantTier: ConsultantTier
): { platformFee: number; consultantAmount: number; feePct: number } {
  const feePct = consultantTier === 'pro' ? PRICING.FEE_PRO : PRICING.FEE_FREE
  const platformFee = Math.round(grossCents * feePct / 100)
  const consultantAmount = grossCents - platformFee
  return { platformFee, consultantAmount, feePct }
}

/** Formats ZAR cents as R4,500 */
export function formatZAR(cents: number): string {
  return 'R' + (cents / 100).toLocaleString('en-ZA', {
    minimumFractionDigits: 0,
    maximumFractionDigits: 0
  })
}

/** Pro tier break-even: monthly earnings comparison */
export function proBreakEven(filesPerMonth: number) {
  const freeEarnings = (filesPerMonth * (PRICING.FILE_STANDARD - Math.round(PRICING.FILE_STANDARD * PRICING.FEE_FREE / 100)))
  const proEarnings  = (filesPerMonth * (PRICING.FILE_STANDARD - Math.round(PRICING.FILE_STANDARD * PRICING.FEE_PRO   / 100))) - PRICING.PRO_MONTHLY
  return {
    freeEarnings,
    proEarnings,
    advantage: proEarnings - freeEarnings,
    proIsBetter: proEarnings > freeEarnings,
  }
}

/** Determines SLA deadline from when payment is confirmed */
export function getSlaDeadline(paidAt: Date): Date {
  const deadline = new Date(paidAt)
  deadline.setHours(deadline.getHours() + PRICING.SLA_HOURS)
  return deadline
}

/** Selects the correct template and returns everything needed to generate a draft */
export async function selectTemplate(answers: WizardAnswers) {
  const code = PROJECT_TYPE_MAP[answers.projectType]
  if (!code) throw new Error(`Unknown project type: ${answers.projectType}`)

  const complex = isComplex(answers)
  const supabase = createAdminClient()

  const { data, error } = await supabase
    .from('project_types')
    .select(`
      id, code, label, requires_chsm, high_risk, price_standard, price_complex,
      she_templates (
        id, content, version,
        customisation_minutes_min, customisation_minutes_max
      )
    `)
    .eq('code', code)
    .single()

  if (error || !data) throw new Error(`Project type not found: ${code}`)

  const templates = (data as any).she_templates as any[]
  const template  = templates?.find((t: any) => t.is_active !== false) ?? templates?.[0]

  return {
    projectTypeId:     data.id,
    templateId:        template?.id ?? null,
    templateContent:   template?.content ?? null,
    templateVersion:   template?.version ?? '1.0',
    isComplex:         complex,
    requiresChsm:      complex || data.requires_chsm,
    price:             complex ? data.price_complex : data.price_standard,
    customisationMins: {
      min: template?.customisation_minutes_min ?? 30,
      max: template?.customisation_minutes_max ?? 45,
    },
  }
}
